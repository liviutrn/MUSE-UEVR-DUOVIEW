cmake_minimum_required(VERSION 3.20)
project(DuoViewIndependentTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(DuoViewGeometryTests stage2_geometry_tests.cpp)
add_executable(DuoViewProbe DuoViewProbe.cpp)
add_executable(DuoViewGazeTests stage4_gaze_tests.cpp)
add_executable(DuoViewFrameTests stage5_frame_tests.cpp)
add_executable(DuoViewConfigTests stage8_config_tests.cpp)
add_executable(DuoViewStateTests stage10_state_tests.cpp)
if(WIN32)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/DuoViewShaderSource.hpp"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/embed_shader.py" "${CMAKE_CURRENT_BINARY_DIR}/DuoViewShaderSource.hpp"
    DEPENDS embed_shader.py DuoViewComposite.hlsl DuoViewCompositeRaster.hlsl DuoViewNativeCompositeRaster.hlsl)
  add_executable(DuoViewProductionWarpTests production_warp_test.cpp "${CMAKE_CURRENT_BINARY_DIR}/DuoViewShaderSource.hpp")
  target_include_directories(DuoViewProductionWarpTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(DuoViewProductionWarpTests PRIVATE d3d12 dxgi d3dcompiler)
  add_executable(DuoViewWarpTests DuoViewWarpTest.cpp)
  target_link_libraries(DuoViewWarpTests PRIVATE d3d12 dxgi d3dcompiler)
endif()
enable_testing()
add_test(NAME geometry COMMAND DuoViewGeometryTests)
add_test(NAME gaze COMMAND DuoViewGazeTests)
add_test(NAME frame COMMAND DuoViewFrameTests)
add_test(NAME config COMMAND DuoViewConfigTests)

add_test(NAME state COMMAND DuoViewStateTests)
foreach(test DuoViewGeometryTests DuoViewGazeTests DuoViewFrameTests DuoViewConfigTests DuoViewStateTests)
  if(MSVC)
    target_compile_options(${test} PRIVATE /UNDEBUG)
  else()
    target_compile_options(${test} PRIVATE -UNDEBUG)
  endif()
endforeach()
if(WIN32)
  add_test(NAME production-warp COMMAND DuoViewProductionWarpTests)
  set_tests_properties(production-warp PROPERTIES TIMEOUT 60)
endif()
